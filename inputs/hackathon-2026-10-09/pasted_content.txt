const [busy, setBusy] = useState<"csv" | "package" | "decripto" | "finalize" | "reissue" | null>(null);

const reissue = async () => {
  if (!data?.outdated) return;
  setBusy("reissue"); setActionError(null); setNotice(null);
  try {
    const updated = await api.reissueReport(mes);
    setData(updated);                 // attestation volta null → o polling de 5 s já existente recarrega
    setNotice(t.report.reissued(updated.version ?? 2));
  } catch (err) {
    setActionError(errorMessage(err)); // 409 attestation_pending / report_up_to_date
  } finally {
    setBusy(null);
  }
};
